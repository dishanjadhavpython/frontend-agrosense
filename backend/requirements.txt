# The reading service: get twelve numbers off a Soil Health Card, index the
# document for retrieval, run the three prediction models, and run the research
# agents behind the detail pages.
#
# Nothing in `_unwired/` is covered by this file (boto3, Clerk). Adding those
# back is a decision, not an install.
#
# Verified to resolve as arm64 wheels on the project's Python 3.14 venv.

fastapi>=0.115,<1.0
uvicorn[standard]>=0.34,<1.0
python-multipart>=0.0.20         # FastAPI multipart uploads

PyMuPDF>=1.26,<2.0               # PDF text extraction and page rasterising
numpy>=2.0,<3.0
scikit-learn>=1.5,<2.0           # HashingVectorizer / TfidfVectorizer

Pillow>=11.0,<13.0               # image decode + pre-OCR grayscale/autocontrast
pytesseract>=0.3.13,<1.0         # needs the `tesseract` binary on PATH

# HEIC support is OPTIONAL and deliberately not installed by default: the wheel
# builds against libheif, which is a system package, and it fails on Python 3.14
# without it. iOS Safari transcodes HEIC to JPEG for ordinary file uploads, so
# this only matters for a file dragged straight off a Mac's Photos library.
# To enable:  brew install libheif && pip install pillow-heif
# `ingest.py` detects it at import and returns a clear error when it is absent.

httpx>=0.27,<1.0                 # Ollama client (optional answer path)

# --- Prediction models ----------------------------------------------------
# The soil classifier is an EfficientNet-B0 fine-tune (~16 MB); torch itself is
# ~200 MB installed and is the heaviest thing in this file. `backend/config.py`
# pins OMP to one thread at import time, without which torch and scikit-learn
# each load their own libomp and the process segfaults on any request that
# touches both models.
torch>=2.2                       # CPU is fine; no CUDA build required
torchvision>=0.17
xgboost>=2.0,<4.0                # fertilizer model
pandas>=2.0,<4.0                 # fertilizer model expects a named DataFrame

# The crop model, after the bake-off picked LightGBM over XGBoost — and also
# the OpenMP fix. `models.py` imports it before torch on purpose: letting
# LightGBM initialise OpenMP first is what stops torch's own libomp from
# segfaulting the process on any request that touches both. Not optional, and
# not only needed to unpickle the model.
lightgbm>=4.0,<5.0

# --- Research agents ------------------------------------------------------
# Planner -> Research -> Creator -> Reviewer, on a background sweep, feeding
# the insights section of every detail page. `backend/app.py` imports the
# package unconditionally and the lifespan hook imports the scheduler, so
# these are required to *start* the service, not only to run a sweep — the
# subsystem turns itself off by having no OPENAI_API_KEY, not by being absent.
python-dotenv>=1.0,<2.0          # config.py reads .env / .env.local itself
openai-agents>=0.19,<1.0         # the Agent/Runner loop (pulls in `openai`)
mcp>=1.29,<2.0                   # the four stdio tool servers
ddgs>=9.0,<10.0                  # web_search_server's backend
APScheduler>=3.11,<4.0           # the sweep timer
# fetch_server strips a page to readable text with it. It arrives anyway as a
# dependency of ddgs, which is why nothing had noticed it missing — pinned here
# because a direct import should not rest on another package's dependency list.
lxml>=5.0,<7.0
